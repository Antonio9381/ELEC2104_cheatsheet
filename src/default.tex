% ---------- Headings (numbered, listed in contents and PDF bookmarks) ----------
% Optional [number] uses the textbook's numbering; omit it to auto-number.
%   \section[10]{Analog Systems}     -> 10 Analog Systems
%   \subsection[10.4]{Bode Plots}    -> 10.4 Bode Plots
%   \subsubsection{Label}            -> small label, not in contents
\makeatletter
\newdimen\sheet@headw
\newif\ifsheet@afterhead
\newcommand{\sheet@ifblank}[3]{\if\relax\detokenize{#1}\relax #2\else #3\fi}
\renewcommand{\section}[2][]{%
    \par\needspace{6\baselineskip}\vspace{2pt}%
    \sheet@ifblank{#1}{}{\setcounter{section}{\numexpr#1-1\relax}}%
    \global\sheet@afterheadfalse
    \refstepcounter{section}%
    \addcontentsline{toc}{section}{\protect\numberline{\thesection}#2}%
    \noindent\begin{tcolorbox}[sheethead, colback=secLight,
        borderline west={2.5pt}{0pt}{secDark}, top=0.6mm, bottom=0.6mm]
        \scriptsize\bfseries\color{secDark}\thesection\enspace #2
    \end{tcolorbox}\par
    \vspace{1pt}%
}
\renewcommand{\subsection}[2][]{%
    \par\needspace{5\baselineskip}\vspace{2pt}%
    \refstepcounter{subsection}%
    \sheet@ifblank{#1}{\edef\@subnum{\thesubsection}}{\def\@subnum{#1}}%
    \addcontentsline{toc}{subsection}{\protect\numberline{\@subnum}#2}%
    \sbox0{\tcbox[sheethead, colback=subLight, borderline west={2pt}{0pt}{subDark}]{%
        \scriptsize\bfseries\@subnum\enspace #2}}%
    \ifdim\wd0>\linewidth   % too long: wrap onto two lines
        \sbox0{\tcbox[sheethead, colback=subLight, borderline west={2pt}{0pt}{subDark}]{%
            \parbox{\dimexpr\linewidth-2mm-1pt}{\raggedright\scriptsize\bfseries\@subnum\enspace #2}}}%
    \fi
    \global\sheet@headw=\wd0 \global\sheet@afterheadtrue
    \noindent\usebox0\par
}
\renewcommand{\subsubsection}[1]{%
    \par\needspace{4\baselineskip}\vspace{1pt}%
    \global\sheet@afterheadfalse
    \noindent\tcbox[sheethead, colback=subsubLight, borderline west={1.5pt}{0pt}{subsubDark},
        top=0.2mm, bottom=0.2mm]{\tiny\bfseries\color{subsubDark!70!black}#1}\par
}

% Compact table of contents: \contents
\newcommand{\contents}{%
    \noindent\begin{tcolorbox}[sheethead, colback=gray!15,
        borderline west={2.5pt}{0pt}{gray!70!black}, top=0.6mm, bottom=0.6mm]
        \scriptsize\bfseries Contents
    \end{tcolorbox}\par
    \begingroup
        \renewcommand{\l@section}[2]{\par\noindent\textbf{##1}\dotfill\textbf{##2}}%
        \renewcommand{\l@subsection}[2]{\par\noindent\hspace*{1em}##1\dotfill##2}%
        \renewcommand{\numberline}[1]{\makebox[2em][l]{##1}}%
        \@starttoc{toc}%
    \endgroup
    \par\vspace{2pt}%
}
\makeatother

% Textbook reference:  \tb{p.152} or \tb{4.3}  -> grey [J p.152], right-aligned.
% Straight after \subsection it sits on the heading's line if there is room,
% otherwise on its own line. Mid-paragraph it goes at the end of the line.
\makeatletter
\newcommand{\tb}[1]{%
    \sbox2{\color{gray}[J~#1]}%
    \ifvmode
        \ifsheet@afterhead
            \ifdim\dimexpr\sheet@headw+\wd2+0.5em\relax<\linewidth
                \vspace{-\baselineskip}%
            \fi
        \fi
        \noindent\hfill\usebox2\par
    \else
        \hfill\usebox2%
    \fi
    \global\sheet@afterheadfalse}
\makeatother

% Tiny ideal frequency-response sketch:  \filtresp{Title}{curve path}{extra tikz}
%   e.g. \filtresp{LP}{(0,1) -- (0.5,1) -- (0.5,0)}{}   (x from 0 to 1, y from 0 to 1)
\newcommand{\filtresp}[3]{%
    \begin{tikzpicture}[x=0.8cm, y=0.3cm, baseline=0]
        \node[above] at (0.55,1.1) {\tiny\bfseries #1};
        \draw[->] (0,0) -- (1.15,0) node[right, inner sep=0.5pt] {\tiny$f$};
        \draw[->] (0,0) -- (0,1.3);
        \draw[thick, subsubDark] #2;
        #3
    \end{tikzpicture}}
